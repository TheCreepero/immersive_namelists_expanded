# Division template historical names system for Spain (SPR).
# Immersive Namelists Expanded (INEX)
# Spanish ordinals: 1.ª (feminine), 1.º and 1.er (masculine). Honorifics are in single quotes. The Basque and Catalan lists keep their own spelling.
# Overrides all 21 vanilla groups (SPR_*, SPA_INF_01, SPD_INF_01 to SPD_INF_06). Every group is for_countries = { SPR }, as in vanilla.

# ===== Field divisions =====

# Overrides vanilla SPR_INF_01 as the plain variant, with vanilla's gate. The People's Army numbered its divisions without an arm (11.ª División).
# Vanilla's civil-war, Carabineros and Chilean templates use this tag.

{
	name = "Republican Infantry"

	for_countries = { SPR }

	can_use = { NOT = { has_government = fascism } }

	division_types = { "infantry" }

	fallback_name = "%d.ª División"
}
